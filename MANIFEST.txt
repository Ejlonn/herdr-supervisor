# Reviewed source-to-install mapping. User configuration and state are never sourced from live files.
# The program files form one immutable generation; ~/.local/lib/herdr-supervisor/active is the atomic pointer to it.
src/*.py -> ~/.local/lib/herdr-supervisor/generations/<version>-<program id>/
config/quota-dependency.json -> ~/.local/lib/herdr-supervisor/generations/<version>-<program id>/
bin/* -> ~/.local/bin/
systemd/* -> ~/.config/systemd/user/
config/*.example.json -> ~/.local/share/doc/herdr-supervisor/examples/
docs/*.md -> ~/.local/share/doc/herdr-supervisor/
README.md LICENSE VERSION MANIFEST.txt -> ~/.local/share/doc/herdr-supervisor/
